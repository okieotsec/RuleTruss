#!/bin/bash
# usage: loop.sh <target> <seconds>: run the target; after a crash keep the artifact and start again until the time is up
t=$1; end=$(( $(date +%s) + $2 )); n=0
while [ $(date +%s) -lt $end ]; do
  n=$((n+1)); left=$(( end - $(date +%s) ))
  RT_REPO="$REPO" PYTHONPATH="$REPO" "$PYTHON_WITH_ATHERIS" fuzz.py $t -max_total_time=$left -timeout=30 -rss_limit_mb=1024 -max_len=65536 -artifact_prefix=crashes/$t-r$n- corpus/$t > run-$t-r$n.log 2>&1
  ls crashes/$t-r$n-* >/dev/null 2>&1 || break
done
echo finished
